%=================================================== (b) the two models
\begin{scope}[xshift=8.15cm, yshift=1.60cm]

  \begin{scope}[xshift=\dx cm]
    % wall + one mirror element
    \draw[black!45, line width=0.6pt] (0,-1.85) -- (0,1.75);
    \fill[black!10] (-0.16,-1.85) rectangle (0,1.75);
